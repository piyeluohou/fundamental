import React, { useState, useEffect, useRef } from "react";
import ReactDOM from "react-dom";
function Counter() {
 const [count, setCount] = useState(0);
 useEffect(() => {
  setInterval(() => {
   setCount(count + 1);
  }, 1000);
 }, []);
 return <h1>{count}</h1>;
}
const rootElement = document.getElementById("root");
ReactDOM.render(<Counter />, rootElement);
